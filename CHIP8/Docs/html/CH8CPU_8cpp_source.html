37void CH8CPU::fetch() {
38 // 1. Core Byte Assembly Extraction
39 Uint8 highByte = readBus(PC);
40 Uint8 lowByte = readBus(PC + 1);
41 opc = (highByte << 8) | lowByte;
42
43 // 2. Opcode Matrix Lookup Identification moved here
44 opcode = nullptr;
45 for (int i = 0; i < TABLE_SIZE; ++i) {
46 if ((opc & opcodes[i].mask) == opcodes[i].id) {
47 opcode = &opcodes[i];
48 break;
49 }

118void CH8CPU::opArithmetic() {
119 Uint8 x = (opc & 0x0F00) >> 8;
120 Uint8 y = (opc & 0x00F0) >> 4;
121
122 switch (opc & 0x000F) {
123 case 0x0:
124 V[x] = V[y];
125 break;
126
127 case 0x1:
128 V[x] |= V[y];
129 V[0xF] = 0;
130 break;
131
132 case 0x2:
133 V[x] &= V[y];
134 V[0xF] = 0;
135 break;
136
137 case 0x3:
138 V[x] ^= V[y];
139 V[0xF] = 0;
140 break;
141
142 case 0x4: {
143 // ADD Vx, Vy: Vx = Vx + Vy. VF = 1 if carry occurs, 0 otherwise.
144 Uint16 result = (Uint16)V[x] + (Uint16)V[y];
145 Uint8 carry = (result > 255) ? 1 : 0;
146
147 V[x] = result & 0xFF;
148 V[0xF] = carry; // Flag assigned last, preventing premature overwrite
149 break;
150 }
151
152 case 0x5: {
153 // SUB Vx, Vy: Vx = Vx - Vy. VF = 1 if NOT a borrow (Vx >= Vy), 0 otherwise.
154 Uint8 notBorrow = (V[x] >= V[y]) ? 1 : 0;
155 Uint8 result = V[x] - V[y];
156
157 V[x] = result;
158 V[0xF] = notBorrow; // Flag assigned last
159 break;
160 }
161
162 case 0x6: {
163 // SHIFTING ON: Vx = Vy >> 1. VF = dropped LSB.
164 Uint8 lsb = V[y] & 0x01;
165 V[x] = V[y] >> 1;
166 V[0xF] = lsb;
167 break;
168 }
169
170 case 0x7: {
171 // SUBN Vx, Vy: Vx = Vy - Vx. VF = 1 if NOT a borrow (Vy >= Vx), 0 otherwise.
172 Uint8 notBorrow = (V[y] >= V[x]) ? 1 : 0;
173 Uint8 result = V[y] - V[x];
174
175 V[x] = result;
176 V[0xF] = notBorrow; // Flag assigned last
177 break;
178 }
179
180 case 0xE: {
181 // SHIFTING ON: Vx = Vy << 1. VF = dropped MSB.
182 Uint8 msb = (V[y] & 0x80) >> 7;
183 V[x] = V[y] << 1;
184 V[0xF] = msb;
185 break;
186 }
187
188 default:
189 break;
190 }
191}

212void CH8CPU::opTimersAndMemory() {
213 Uint8 x = (opc & 0x0F00) >> 8;
214 switch (opc & 0x00FF) {
215 case 0x07: V[x] = DELAY_TIMER; break;
216 case 0x15: DELAY_TIMER = V[x]; break;
217 case 0x18: SOUND_TIMER = V[x]; break;
218 case 0x1E: I += V[x]; break;
219 case 0x0A: {
220 // Stage 1: No key has been locked down yet
221 if (!emu->key.isCurrentlyWaiting()) {
222 for (Uint8 i = 0; i < 16; ++i) {
223 if (emu->key.isKeyPressed(i)) {
224 // Delegate tracking management directly to the controller
225 emu->key.lockKeyWait(i);
226 break;
227 }
228 }
229 // Halt effect: Keep PC pointing here until a key down-press occurs
230 PC -= 2;
231 }
232 // Stage 2: Controller is tracking a held key; wait for release
233 else {
234 Uint8 activeKey = static_cast<Uint8>(emu->key.getLatchedKey());
235
236 if (!emu->key.isKeyPressed(activeKey)) {
237 // Success! The physical key has transitioned to a released state
238 V[x] = activeKey;
239 emu->key.clearKeyWait();
240 // Do NOT decrement PC here; execution advances normally
241 } else {
242 // Keep halting: Key is still held down
243 PC -= 2;
244 }
245 }
246 break;
247 }
248 case 0x29: I = V[x] * 5; break;
249 case 0x33: writeBus(I, V[x]/100); writeBus(I+1, (V[x]/10)%10); writeBus(I+2, V[x]%10); break;
250
251 case 0x55:
252 for (int i = 0; i <= x; ++i) {
253 writeBus(I + i, V[i]);
254 }
255 I += x + 1;
256 break;
257
258 case 0x65:
259 for (int i = 0; i <= x; ++i) {
260 V[i] = readBus(I + i);
261 }
262 I += x + 1;
263 break;
264 }
265}
